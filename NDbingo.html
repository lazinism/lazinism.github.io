const canvas = document.getElementById('bingoCanvas'); const ctx = canvas.getContext('2d'); const inputGrid = document.getElementById('inputGrid'); const updateButton = document.getElementById('updateButton'); const img = new Image(); const gridRows = 5; const gridCols = 5; const numCells = gridRows * gridCols; img.onload = () => { const aspect = img.width / img.height; canvas.width = 800; canvas.height = canvas.width / aspect; createInputs(); drawBingo(); }; // Base64 문자열 대신 로컬 파일 경로 지정 // HTML 파일과 image_61e0e9.jpg 파일이 같은 폴더 위치(루트)에 있어야 정상 작동합니다. img.src = 'etcs/bingo.jpg'; function createInputs() { for (let i = 0; i < numCells; } } function wrapText(context, text, x, y, maxWidth, lineHeight) { const words = text.split(' '); let line = ''; let numLines = 0; for (let n = 0; n < words.length; n++) { const testLine = line + words[n] + ' '; const metrics = context.measureText(testLine); const testWidth = metrics.width; if (testWidth > maxWidth && n > 0) { context.fillText(line, x, y); line = words[n] + ' '; y += lineHeight; numLines++; } else { line = testLine; } } context.fillText(line, x, y); numLines++; return numLines; } function drawBingo() { ctx.clearRect(0, 0, canvas.width, canvas.height); ctx.drawImage(img, 0, 0, canvas.width, canvas.height); const gridAreaStartPercent = 14; const cellStartWidthPercent = 2.5; const gridAreaWidthPercent = 95; const gridAreaHeightPercent = 84; const gridX = canvas.width * (cellStartWidthPercent / 100); const gridY = canvas.height * (gridAreaStartPercent / 100); const gridWidth = canvas.width * (gridAreaWidthPercent / 100); const gridHeight = canvas.height * (gridAreaHeightPercent / 100); const cellWidth = gridWidth / gridCols; const cellHeight = gridHeight / gridRows; const bgAlpha = 0.6; const textColor = '#FF0000'; const fontSize = 18; const lineHeight = 22; const maxWidthPaddingPercent = 10; for (let r = 0; r < gridRows; r++) { for (let c = 0; c < gridCols; c++) { const i = r * gridCols + c; const text = document.getElementById(`input-${i}`).value; if (text) { const x = gridX + c * cellWidth; const y = gridY + r * cellHeight; ctx.fillStyle = `rgba(255, 255, 255, ${bgAlpha})`; const cellPadding = 2; ctx.fillRect(x + cellPadding, y + cellPadding, cellWidth - 2 * cellPadding, cellHeight - 2 * cellPadding); ctx.fillStyle = textColor; ctx.font = `bold ${fontSize}px sans-serif`; ctx.textAlign = 'center'; ctx.textBaseline = 'middle'; const textX = x + cellWidth / 2; const textY = y + cellHeight / 2; const maxWidth = cellWidth * (1 - 2 * (maxWidthPaddingPercent / 100)); const linesToCalculateHeight = text.split(' ').length; const estimatedTextHeight = linesToCalculateHeight * lineHeight; const startingY = textY - estimatedTextHeight / 2 + lineHeight / 2; wrapText(ctx, text, textX, startingY, maxWidth, lineHeight); } } } } updateButton.addEventListener('click', drawBingo);
